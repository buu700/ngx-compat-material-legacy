const root=process.argv[2],input=process.argv[3],out=process.argv[4],report=JSON.parse(readFileSync(input));const {mixinArgumentResultsMatch}=await import(pathToFileURL(join(root,'scripts/sass-mixin-arguments.mjs')));const sha=s=>createHash('sha256').update(s).digest('hex'),rows=[];
for(const row of report.cases){assert.equal(mixinArgumentResultsMatch(row.expected,row.actual),true);for(const [kind,css]of [['current-peer-css-over-historical',row.peer.css],['unrelated-owned-css',row.actual.css+'\n.unrelated-owned-regression {padding:0;}']]){assert.equal(mixinArgumentResultsMatch(row.expected,{css,css_sha256:sha(css),css_bytes:Buffer.byteLength(css)}),false);rows.push({case_id:row.case_id,kind,rejected:true});}}
